Repository root:
  $HOME/Documents/Thesis git/drone_cage_control

1. Replace the standalone tool with this cumulative B.4.2 package

  cd "$HOME/Documents/Thesis git/drone_cage_control"
  rm -rf tools/standalone_dynamic_planner_cpp
  unzip "$HOME/Downloads/R6_3B_4_suspended_system_aug31_v3.zip" -d tools/

2. Configure and build

  cd "$HOME/Documents/Thesis git/drone_cage_control/tools/standalone_dynamic_planner_cpp"
  rm -rf build
  cmake -S . -B build \
    -G Ninja \
    -DCMAKE_BUILD_TYPE=Release \
    -DCMAKE_PREFIX_PATH="$HOME/.local"
  cmake --build build -j

  ctest --test-dir build --output-on-failure
